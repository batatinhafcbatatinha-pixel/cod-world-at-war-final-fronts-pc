cmake_minimum_required(VERSION 3.25)
project(CodWawFinalFrontsNative LANGUAGES C CXX)
get_filename_component(WORK_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
# PS2Recomp runtime: the third_party/PS2Recomp submodule (fork, branch native-ports) when checked out,
# otherwise the shared checkout next to the game folders (../third_party/PS2Recomp).
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../third_party/PS2Recomp/CMakeLists.txt")
    set(CODWAW_PS2RECOMP_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../third_party/PS2Recomp")
else()
    set(CODWAW_PS2RECOMP_DIR "${WORK_ROOT}/third_party/PS2Recomp")
endif()
get_filename_component(CODWAW_PS2RECOMP_DIR "${CODWAW_PS2RECOMP_DIR}" ABSOLUTE)
# The game's private supplements (codwaw_*_chunk_*.cpp, which include generated game code) exist only in the
# author's shared checkout; the published fork leaves them out.
if(EXISTS "${CODWAW_PS2RECOMP_DIR}/ps2xRuntime/src/lib/codwaw_gaps_chunk_00.cpp")
    set(PS2X_ENABLE_CODWAW_SUPPLEMENTS ON CACHE BOOL "CoD game bindings")
else()
    set(PS2X_ENABLE_CODWAW_SUPPLEMENTS OFF CACHE BOOL "CoD game bindings")
endif()
set(PS2X_RUNNER_DIR "${CMAKE_CURRENT_SOURCE_DIR}/runner" CACHE PATH "CoD AOT sources")
set(PS2X_GAME_CHUNK_DIR "${CMAKE_CURRENT_SOURCE_DIR}/runner-chunks" CACHE PATH "CoD AOT chunks")
set(PS2X_GAME_LIB_PREBUILT "" CACHE FILEPATH "Optional native archive for this platform")
set(PS2X_BUILD_RECOMP OFF CACHE BOOL "Runtime only")
set(PS2X_BUILD_ANALYZER OFF CACHE BOOL "Runtime only")
set(PS2X_BUILD_TEST OFF CACHE BOOL "No tests")
set(PS2X_BUILD_STUDIO OFF CACHE BOOL "Runtime only")
set(PS2X_ENABLE_DEBUG_UI OFF CACHE BOOL "Game frontend")
set(PS2X_ENABLE_LTCG OFF CACHE BOOL "Bound build memory")
set(PS2X_REQUIRE_GPU_RASTERIZER ON CACHE BOOL "Native host GPU")
add_subdirectory("${CODWAW_PS2RECOMP_DIR}" ps2recomp)

# Shared data installer: extracted files back all native EE/IOP optical reads.
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../tools/port-installer/runtime/extracted_files.cmake")
    set(PORT_INSTALLER_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../tools/port-installer" CACHE PATH "Shared installer checkout")
else()
    set(PORT_INSTALLER_DIR "${WORK_ROOT}/port-installer" CACHE PATH "Shared installer checkout")
endif()
include("${PORT_INSTALLER_DIR}/runtime/extracted_files.cmake")
